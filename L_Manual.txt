local function getGradientSequence(left, right)
    left = left or Color3.fromRGB(255,255,255)
    right = right or Color3.fromRGB(255,255,255)
    local midColor = Color3.fromRGB(
        math.clamp((left.R + right.R) * 0.5 * 255, 0, 255),
        math.clamp((left.G + right.G) * 0.5 * 255, 0, 255),
        math.clamp((left.B + right.B) * 0.5 * 255, 0, 255)
    )
    return ColorSequence.new({
        ColorSequenceKeypoint.new(0, left),
        ColorSequenceKeypoint.new(0.5, midColor),
        ColorSequenceKeypoint.new(1, right)
    })
end

local function getAdaptiveTextColor(bgLeft, bgRight)
    bgLeft = bgLeft or Color3.fromRGB(0,0,0)
    bgRight = bgRight or Color3.fromRGB(0,0,0)
    local avgLuminance = ((bgLeft.R + bgLeft.G + bgLeft.B) / 3 + (bgRight.R + bgRight.G + bgRight.B) / 3) / 2
    if avgLuminance > 0.73 then
        return Color3.fromRGB(20, 20, 25)
    else
        return Color3.fromRGB(255, 255, 255)
    end
end

local function GetBallSpeed()

local function GetPing()
    local ok, result = pcall(function()
        return StatsService.Network.ServerStatsItem["Data Ping"]:GetValue()
    end)
    return ok and result or 100
end

local Lerp_Radians = 0
local Last_Warping = tick()

local function Is_Curved(ball)
    local Zoomies = ball:FindFirstChild("zoomies")
    if not Zoomies then return false end
    local Velocity = Zoomies.VectorVelocity
    local Character = player.Character
    if not Character or not Character.PrimaryPart then return false end

    local distanceToBall = (Character.PrimaryPart.Position - ball.Position).Magnitude
    if distanceToBall <= 25 then return false end

    local Speed = Velocity.Magnitude
    local Direction = (Character.PrimaryPart.Position - ball.Position).Unit
    local Dot = Direction:Dot(Velocity.Unit)
    local Ping = GetPing() / 1000
    local Distance = (Character.PrimaryPart.Position - ball.Position).Magnitude
    local Reach_Time = Distance / Speed - Ping
    local Radians = math.rad(math.asin(math.clamp(Dot, -1, 1)))
    Lerp_Radians = Lerp_Radians + (Radians - Lerp_Radians) * 0.8

    if Lerp_Radians < 0.018 then Last_Warping = tick() end
    if (tick() - Last_Warping) < (Reach_Time / 1.5) then return true end
    return Dot < (0.5 - Ping)
end

local function GetParryAnimation()
    local char = player.Character
    local currentSword = char and char:GetAttribute("CurrentlyEquippedSword")
    if not currentSword then
        return SwordAPI.Collection.Default:FindFirstChild("GrabParry")
    end
    if AnimationCache[currentSword] then return AnimationCache[currentSword] end
    local ok, swordData = pcall(function()
        return ReplicatedStorage.Shared.ReplicatedInstances.Swords.GetSword:Invoke(currentSword)
    end)
    if ok and type(swordData) == "table" then
        for _, obj in pairs(SwordAPI.Collection:GetChildren()) do
            if obj.Name == swordData.AnimationType then
                local anim = obj:FindFirstChild("GrabParry") or obj:FindFirstChild("Grab")
                if anim then
                    AnimationCache[currentSword] = anim
                    return anim
                end
            end
        end
    end
    return SwordAPI.Collection.Default:FindFirstChild("GrabParry")
end

local function PlayParryAnimation()
    if not Config.AnimFix then return end
    local hum = player.Character and player.Character:FindFirstChildOfClass("Humanoid")
    if not hum or not hum:FindFirstChild("Animator") then return end
    local animation = GetParryAnimation()
    if not animation then return end
    for _, track in pairs(hum.Animator:GetPlayingAnimationTracks()) do
        if track.Name:find("Grab") or track.Name:find("Parry") then
            track:Stop(0.1)
        end
    end
    local track = hum.Animator:LoadAnimation(animation)
    track:Play(0, 1, 1)
end

local function ApplyCurveToCFrame(baseCFrame)
    if not Config.CurveType or Config.CurveType == "straight" then return baseCFrame end
    local rotation = CFrame.new()
    if Config.CurveType == "backwards" then
        rotation = CFrame.Angles(0, math.rad(180), 0)
    elseif Config.CurveType == "down" then
        rotation = CFrame.Angles(math.rad(-45), 0, 0)
    elseif Config.CurveType == "up" then
        rotation = CFrame.Angles(math.rad(45), 0, 0)
    elseif Config.CurveType == "left" then
        rotation = CFrame.Angles(0, math.rad(-90), 0)
    elseif Config.CurveType == "right" then
        rotation = CFrame.Angles(0, math.rad(90), 0)
    elseif Config.CurveType == "random" then
        rotation = CFrame.Angles(0, math.random() * math.pi * 2, 0)
    end
    return baseCFrame * rotation
end

local function SendParry()
    if not ParryRemote then return false end
    local Hash1 = getgenv()._Kittylol_Hash1
    local Hash2 = getgenv()._Kittylol_Hash2
    local Hash3 = getgenv()._Kittylol_Hash3
    if not Hash1 or not Hash2 or not Hash3 then return false end

    local camCF, events, mousePos = GetParryData()
    local modifiedCF = ApplyCurveToCFrame(camCF)
    local success = pcall(function()
        ParryRemote:FireServer(
            Hash1,
            Hash2,
            Hash3(),
            0.025,
            modifiedCF,
            events,
            mousePos,
            false
        )
    end)
    if success and Config.AnimFix then
        task.spawn(PlayParryAnimation)
    end
    return success
end

local function ProcessAutoParry(ball)
    if not Config.AutoParry or spamActive then return end
    local bID = ball:GetDebugId()
    if ball:GetAttribute("target") ~= player.Name or parried_balls[bID] then return end
    if Is_Curved(ball) then return end
    local charPart = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    if not charPart then return end
    local velocity = ball.zoomies.VectorVelocity
    local dist = (charPart.Position - ball.Position).Magnitude
    local ping = GetPing()
    local threshold = (velocity.Magnitude / Config.ParryAccuracy) + (ping / 10)
    if dist <= threshold or dist <= 20 then
        parried_balls[bID] = true
        SendParry()
        ball:GetAttributeChangedSignal("target"):Once(function()
            parried_balls[bID] = nil
        end)
    end
end

local function ProcessTriggerBot(ball)
    if not Config.Trigger or spamActive then return end
    local bID = ball:GetDebugId()
    if ball:GetAttribute("target") ~= player.Name and not triggered_balls[bID] then
        triggered_balls[bID] = true
        SendParry()
        ball:GetAttributeChangedSignal("target"):Once(function()
            triggered_balls[bID] = nil
        end)
    end
end

local function MakeDraggable(frame)
    local dragging, dragInput, dragStart, startPos
    frame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    frame.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UIS.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

local function BuildKeyRow(parent, keys, yPos, indent)
    local row = Instance.new("Frame", parent)
    row.BackgroundTransparency = 1
    row.Size = UDim2.new(1, -16, 0, 30)
    row.Position = UDim2.new(0, 8, 0, yPos)
    local count = #keys
    local spacing = 4
    local totalW = 424 - indent * 2
    local keyW = (totalW - (count - 1) * spacing) / count
    for i, keyData in ipairs(keys) do
        local display, code = keyData[1], keyData[2]
        local btn = Instance.new("TextButton", row)
        btn.Size = UDim2.new(0, keyW, 1, 0)
        btn.Position = UDim2.new(0, indent + (i - 1) * (keyW + spacing), 0, 0)
        btn.BackgroundColor3 = Color3.fromRGB(35, 35, 45)
        btn.Text = display
        btn.TextColor3 = Color3.fromRGB(255, 255, 255)
        btn.Font = Enum.Font.GothamBold
        btn.TextSize = 13
        btn.ZIndex = 6
        btn.BorderSizePixel = 0
        Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 5)
        kbKeys[code] = btn
        btn.MouseEnter:Connect(function()
            if code ~= spamBindKey then btn.BackgroundColor3 = Color3.fromRGB(55, 55, 68) end
        end)
        btn.MouseLeave:Connect(function()
            RefreshKeyboardHighlight()
        end)
        btn.MouseButton1Click:Connect(function()
            spamBindKey = code
            if BindButton then BindButton.Text = "[ " .. spamBindKey .. " ]" end
            RefreshKeyboardHighlight()
            KeyboardFrame.Visible = false
            Config.ShowBindWindow = false
        end)
    end
end

local IsMinimized = false
local IsTransitioning = false
local PreMinimizeSize = MainFrame.Size
local PreMinimizeChildren = {}

local function ConnectDraggingAndMinimize(frame, dragButton)
    local activeInput, dragStart, startPos, touchStartTime, touchStartPos
    local isDragging = false

    dragButton.InputBegan:Connect(function(input)
        if (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) and not activeInput then
            activeInput = input
            dragStart = input.Position
            startPos = frame.Position
            touchStartTime = os.clock()
            touchStartPos = input.Position
            isDragging = false
        end
    end)

    UIS.InputChanged:Connect(function(input)
        if input == activeInput and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            if delta.Magnitude > 5 then
                isDragging = true
                frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
            end
        end
    end)

    UIS.InputEnded:Connect(function(input)
        if input == activeInput then
            local timeDelta = os.clock() - touchStartTime
            local distanceDelta = (input.Position - touchStartPos).Magnitude
            activeInput = nil

            if not isDragging and distanceDelta < 7 and timeDelta < 0.35 then
                if IsTransitioning then return end
                IsTransitioning = true
                IsMinimized = not IsMinimized

                if IsMinimized then
                    PreMinimizeSize = frame.Size
                    for _, child in pairs(frame:GetChildren()) do
                        if child ~= LeftMenu and child ~= BackgroundImage and child:IsA("GuiObject") then
                            PreMinimizeChildren[child] = child.Visible
                            child.Visible = false
                        end
                    end
                    for _, child in pairs(LeftMenu:GetChildren()) do
                        if child ~= TitleHeader and child:IsA("GuiObject") then
                            PreMinimizeChildren[child] = child.Visible
                            child.Visible = false
                        end
                    end

                    TitleHeader.Size = UDim2.new(1, 0, 0, 24)
                    AnimatedImage.Position = UDim2.new(0, 6, 0, 4)
                    AnimatedImage.Size = UDim2.new(0, 16, 0, 16)
                    NameLabel.Position = UDim2.new(0, 28, 0, 0)
                    NameLabel.Size = UDim2.new(1, -28, 1, 0)
                    NameLabel.TextSize = 11
                    LeftMenu.BackgroundTransparency = 1
                    BackgroundImage.Visible = true
                    UICorner_Main.CornerRadius = UDim.new(0, 6)
                    UICorner_BgImage.CornerRadius = UDim.new(0, 6)

                    local closeTween = TS:Create(frame, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Size = UDim2.new(0, 110, 0, 36)})
                    closeTween:Play()
                    closeTween.Completed:Connect(function()
                        LeftMenu.Size = UDim2.new(0, 98, 0, 24)
                        IsTransitioning = false
                    end)
                else
                    TitleHeader.Size = UDim2.new(1, 0, 0, 44)
                    AnimatedImage.Position = UDim2.new(0, 8, 0, 10)
                    AnimatedImage.Size = UDim2.new(0, 24, 0, 24)
                    NameLabel.Position = UDim2.new(0, 38, 0, 0)
                    NameLabel.Size = UDim2.new(1, -38, 1, 0)
                    NameLabel.TextSize = 13
                    LeftMenu.BackgroundTransparency = 0.88
                    LeftMenu.Size = UDim2.new(0, 130, 0, 308)
                    UICorner_Main.CornerRadius = UDim.new(0, 10)
                    UICorner_BgImage.CornerRadius = UDim.new(0, 10)

                    local openTween = TS:Create(frame, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Size = PreMinimizeSize})
                    openTween:Play()
                    openTween.Completed:Connect(function()
                        for child, wasVisible in pairs(PreMinimizeChildren) do
                            if child and child.Parent then child.Visible = wasVisible end
                        end
                        PreMinimizeChildren = {}
                        for name, frameItem in pairs(TabContainers) do
                            frameItem.Visible = (name == CurrentTab)
                        end
                        IsTransitioning = false
                    end)
                end
            end
        end
    end)
end

        if isTransparency then return tostring(math.floor(v * 100)) .. "%" end
        if decimals then return string.format("%." .. decimals .. "f", v) end
        return tostring(math.floor(v + 0.5))
    end
    SliderValue.Text = formatVal(Config[configKey])

    local SliderBtn = Instance.new("TextButton")
    SliderBtn.Parent = Frame

    local SliderFill = Instance.new("Frame")
    SliderFill.Parent = SliderBtn
    SliderFill.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    SliderFill.BorderSizePixel = 0
    Instance.new("UICorner", SliderFill).CornerRadius = UDim.new(1, 0)
    applyThemeGradient(SliderFill)

    local function getRatio()
        if isTransparency then return Config[configKey] / 0.40 end
        return (Config[configKey] - min) / (max - min)
    end
    SliderFill.Size = UDim2.new(math.clamp(getRatio(), 0, 1), 0, 1, 0)

    local dragging = false
    local function adjustValue(input)
        if SliderBtn.AbsoluteSize.X == 0 then return end
        local posX = math.clamp((input.Position.X - SliderBtn.AbsolutePosition.X) / SliderBtn.AbsoluteSize.X, 0, 1)
        SliderFill.Size = UDim2.new(posX, 0, 1, 0)
        if isTransparency then
            local val = posX * 0.40
            Config[configKey] = val
            MainFrame.BackgroundTransparency = val
            if BackgroundImage then BackgroundImage.ImageTransparency = val end
            SliderValue.Text = tostring(math.floor((val / 0.40) * 100)) .. "%"
        else
            local val = min + (max - min) * posX
            if not decimals then val = math.floor(val + 0.5) end
            Config[configKey] = val
            SliderValue.Text = formatVal(val)
        end
    end

    SliderBtn.MouseButton1Down:Connect(function() dragging = true end)
    UIS.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            adjustValue(input)
        end
    end)
    UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

function UI:CreateAdvancedColorPicker(tabName, mode)
    local targetFrame = TabContainers[tabName]
    if not targetFrame then return end
    local container = Instance.new("Frame")
    container.Size = UDim2.new(1, -14, 0, 110)
    container.BackgroundTransparency = 1
    container.Parent = targetFrame

    local currentSelectedSide = "Left"
    local currentH, currentS, currentV = 0, 1, 1

    local switchRow = Instance.new("Frame")
    switchRow.Size = UDim2.new(1, 0, 0, 20)
    switchRow.BackgroundTransparency = 1
    switchRow.Parent = container

    local leftBtn = Instance.new("TextButton")
    leftBtn.Size = UDim2.new(0.5, -3, 1, 0)
    leftBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 52)
    leftBtn.Text = "Side A"
    leftBtn.TextColor3 = (mode == "Title") and titleColorLeft or themeLeft
    leftBtn.Font = Enum.Font.GothamBold
    leftBtn.TextSize = 10
    leftBtn.Parent = switchRow
    leftBtn.BorderSizePixel = 0
    Instance.new("UICorner", leftBtn).CornerRadius = UDim.new(0, 4)

    local rightBtn = Instance.new("TextButton")
    rightBtn.Size = UDim2.new(0.5, -3, 1, 0)
    rightBtn.Position = UDim2.new(0.5, 3, 0, 0)
    rightBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 52)
    rightBtn.Text = "Side B"
    rightBtn.TextColor3 = (mode == "Title") and titleColorRight or themeRight
    rightBtn.Font = Enum.Font.GothamBold
    rightBtn.TextSize = 10
    rightBtn.Parent = switchRow
    rightBtn.BorderSizePixel = 0
    Instance.new("UICorner", rightBtn).CornerRadius = UDim.new(0, 4)

    local canvas = Instance.new("TextButton")
    canvas.Size = UDim2.new(1, 0, 0, 45)
    canvas.Position = UDim2.fromOffset(0, 26)
    canvas.BackgroundColor3 = Color3.fromHSV(currentH, 1, 1)
    canvas.Text = ""
    canvas.AutoButtonColor = false
    canvas.Parent = container
    canvas.BorderSizePixel = 0
    Instance.new("UICorner", canvas).CornerRadius = UDim.new(0, 4)

    local satGrad = Instance.new("UIGradient")
    satGrad.Color = ColorSequence.new(Color3.fromRGB(255, 255, 255), Color3.fromRGB(0, 0, 0))
    satGrad.Rotation = 90
    satGrad.Parent = canvas

    local valueOverlay = Instance.new("Frame")
    valueOverlay.Size = UDim2.new(1, 0, 1, 0)
    valueOverlay.BackgroundTransparency = 0
    valueOverlay.Parent = canvas
    valueOverlay.BorderSizePixel = 0

    local valGrad = Instance.new("UIGradient")
    valGrad.Color = ColorSequence.new(Color3.fromRGB(255, 255, 255), Color3.fromRGB(0, 0, 0))
    valGrad.Transparency = NumberSequence.new(0, 1)
    valGrad.Parent = valueOverlay

    local pointer = Instance.new("Frame")
    pointer.Size = UDim2.fromOffset(5, 5)
    pointer.Position = UDim2.new(currentS, -2.5, 1 - currentV, -2.5)
    pointer.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    pointer.Parent = canvas
    pointer.BorderSizePixel = 0

    local hueTrack = Instance.new("TextButton")
    hueTrack.Size = UDim2.new(1, 0, 0, 8)
    hueTrack.Position = UDim2.fromOffset(0, 78)
    hueTrack.Text = ""
    hueTrack.AutoButtonColor = false
    hueTrack.Parent = container
    hueTrack.BorderSizePixel = 0
    Instance.new("UICorner", hueTrack).CornerRadius = UDim.new(0, 3)

    local hueGrad = Instance.new("UIGradient")
    hueGrad.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 0, 0)),
        ColorSequenceKeypoint.new(0.166, Color3.fromRGB(255, 255, 0)),
        ColorSequenceKeypoint.new(0.333, Color3.fromRGB(0, 255, 0)),
        ColorSequenceKeypoint.new(0.5, Color3.fromRGB(0, 255, 255)),
        ColorSequenceKeypoint.new(0.666, Color3.fromRGB(0, 0, 255)),
        ColorSequenceKeypoint.new(0.833, Color3.fromRGB(255, 0, 255)),
        ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 0, 0))
    })
    hueGrad.Parent = hueTrack

    local function updatePicker()
        local selectedColor = Color3.fromHSV(currentH, currentS, currentV)
        canvas.BackgroundColor3 = Color3.fromHSV(currentH, 1, 1)
        if mode == "Title" then
            if currentSelectedSide == "Left" then
                titleColorLeft = selectedColor
                leftBtn.TextColor3 = selectedColor
            else
                titleColorRight = selectedColor
                rightBtn.TextColor3 = selectedColor
            end
        else
            if currentSelectedSide == "Left" then
                themeLeft = selectedColor
                leftBtn.TextColor3 = selectedColor
            else
                themeRight = selectedColor
                rightBtn.TextColor3 = selectedColor
            end
        end
        UpdateGlobalTheme()
    end

    leftBtn.MouseButton1Click:Connect(function()
        currentSelectedSide = "Left"
        local baseColor = (mode == "Title") and titleColorLeft or themeLeft
        currentH, currentS, currentV = baseColor:ToHSV()
        pointer.Position = UDim2.new(currentS, -2.5, 1 - currentV, -2.5)
        canvas.BackgroundColor3 = Color3.fromHSV(currentH, 1, 1)
    end)

    rightBtn.MouseButton1Click:Connect(function()
        currentSelectedSide = "Right"
        local baseColor = (mode == "Title") and titleColorRight or themeRight
        currentH, currentS, currentV = baseColor:ToHSV()
        pointer.Position = UDim2.new(currentS, -2.5, 1 - currentV, -2.5)
        canvas.BackgroundColor3 = Color3.fromHSV(currentH, 1, 1)
    end)

    local isSlidingCanvas = false
    canvas.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            isSlidingCanvas = true
            if canvas.AbsoluteSize.X == 0 then return end
            local x = math.clamp((input.Position.X - canvas.AbsolutePosition.X) / canvas.AbsoluteSize.X, 0, 1)
            local y = math.clamp((input.Position.Y - canvas.AbsolutePosition.Y) / canvas.AbsoluteSize.Y, 0, 1)
            pointer.Position = UDim2.new(x, -2.5, y, -2.5)
            currentS = x
            currentV = 1 - y
            updatePicker()
        end
    end)

    UIS.InputChanged:Connect(function(input)
        if isSlidingCanvas and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            if canvas.AbsoluteSize.X == 0 then return end
            local x = math.clamp((input.Position.X - canvas.AbsolutePosition.X) / canvas.AbsoluteSize.X, 0, 1)
            local y = math.clamp((input.Position.Y - canvas.AbsolutePosition.Y) / canvas.AbsoluteSize.Y, 0, 1)
            pointer.Position = UDim2.new(x, -2.5, y, -2.5)
            currentS = x
            currentV = 1 - y
            updatePicker()
        end
    end)

    UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            isSlidingCanvas = false
        end
    end)

    local isSlidingHue = false
    hueTrack.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            isSlidingHue = true
            if hueTrack.AbsoluteSize.X == 0 then return end
            local x = math.clamp((input.Position.X - hueTrack.AbsolutePosition.X) / hueTrack.AbsoluteSize.X, 0, 0.999)
            currentH = x
            updatePicker()
        end
    end)

    UIS.InputChanged:Connect(function(input)
        if isSlidingHue and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            if hueTrack.AbsoluteSize.X == 0 then return end
            local x = math.clamp((input.Position.X - hueTrack.AbsolutePosition.X) / hueTrack.AbsoluteSize.X, 0, 0.999)
            currentH = x
            updatePicker()
        end
    end)

    UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            isSlidingHue = false
        end
    end)
end

function UI:CreateProfileEngine(tabName)
    local targetFrame = TabContainers[tabName]
    if not targetFrame then return end

    local NameInput = Instance.new("TextBox")
    NameInput.Size = UDim2.new(1, -14, 0, 26)
    NameInput.BackgroundColor3 = Color3.fromRGB(35, 35, 45)
    NameInput.Font = Enum.Font.Gotham
    NameInput.PlaceholderText = "Type Profile Name..."
    NameInput.Text = ""
    NameInput.TextColor3 = Color3.fromRGB(255, 255, 255)
    NameInput.PlaceholderColor3 = Color3.fromRGB(120, 120, 130)
    NameInput.TextSize = 11
    NameInput.Parent = targetFrame
    Instance.new("UICorner", NameInput).CornerRadius = UDim.new(0, 5)

    local DropdownBtn = Instance.new("TextButton")
    DropdownBtn.Size = UDim2.new(1, -14, 0, 26)
    DropdownBtn.BackgroundColor3 = Color3.fromRGB(45, 45, 58)
    DropdownBtn.Font = Enum.Font.GothamSemibold
    DropdownBtn.Text = "Select Profile v"
    DropdownBtn.TextColor3 = Color3.fromRGB(200, 200, 210)
    DropdownBtn.TextSize = 11
    DropdownBtn.Parent = targetFrame
    Instance.new("UICorner", DropdownBtn).CornerRadius = UDim.new(0, 5)

    local DropdownListFrame = Instance.new("Frame")
    DropdownListFrame.Size = UDim2.new(1, -14, 0, 0)
    DropdownListFrame.BackgroundColor3 = Color3.fromRGB(30, 30, 38)
    DropdownListFrame.BorderSizePixel = 0
    DropdownListFrame.ClipsDescendants = true
    DropdownListFrame.Parent = targetFrame
    local dropLayout = Instance.new("UIListLayout", DropdownListFrame)
    dropLayout.SortOrder = Enum.SortOrder.LayoutOrder

    local selectedProfileName = ""
    local dropdownOpen = false

    local function rebuildDropdownRows()
        if not DropdownListFrame then return end
        for _, child in ipairs(DropdownListFrame:GetChildren()) do
            if child:IsA("TextButton") then child:Destroy() end
        end
        local profiles = getProfilesList()
        if #profiles == 0 then
            DropdownBtn.Text = "No Profiles Found"
            return
        end
        for _, profName in ipairs(profiles) do
            local row = Instance.new("TextButton")
            row.Size = UDim2.new(1, 0, 0, 22)
            row.BackgroundColor3 = Color3.fromRGB(35, 35, 45)
            row.BackgroundTransparency = 0.4
            row.Font = Enum.Font.Gotham
            row.Text = "  " .. profName
            row.TextColor3 = Color3.fromRGB(230,  230, 230)
            row.MouseButton1Click:Connect(function()
                selectedProfileName = profName
                DropdownBtn.Text = "Selected: " .. profName
                dropdownOpen = false
                TS:Create(DropdownListFrame, TweenInfo.new(0.2), {Size = UDim2.new(1, -14, 0, 0)}):Play()
            end)
        end
    end

    DropdownBtn.MouseButton1Click:Connect(function()
        dropdownOpen = not dropdownOpen
        if dropdownOpen then
            rebuildDropdownRows()
            local totalHeight = dropLayout.AbsoluteContentSize.Y
            TS:Create(DropdownListFrame, TweenInfo.new(0.2), {Size = UDim2.new(1, -14, 0, math.clamp(totalHeight, 0, 100))}):Play()
        else
            TS:Create(DropdownListFrame, TweenInfo.new(0.2), {Size = UDim2.new(1, -14, 0, 0)}):Play()
        end
    end)

    UI:CreateButton("Settings", "Save to Profile", function()
        local configName = NameInput.Text
        if configName == "" then return end

        local currentList = getProfilesList()
        if not table.find(currentList, configName) then
            table.insert(currentList, configName)
            saveProfilesList(currentList)
        end

        local fileTarget = "Kittylol_Profile_" .. configName .. ".json"
        local data = {
            AutoParry = Config.AutoParry,
            ManualSpam = Config.ManualSpam,
            Trigger = Config.Trigger,
            SpamCPS = Config.SpamCPS,
            ParryAccuracy = Config.ParryAccuracy,
            AnimFix = Config.AnimFix,
            CurveType = Config.CurveType,
            UiTransparency = Config.UiTransparency,
            BackgroundId = Config.BackgroundId,
            ThemeLeft = {themeLeft.R, themeLeft.G, themeLeft.B},
            ThemeRight = {themeRight.R, themeRight.G, themeRight.B},
            TitleColorLeft = {titleColorLeft.R, titleColorLeft.G, titleColorLeft.B},
            TitleColorRight = {titleColorRight.R, titleColorRight.G, titleColorRight.B}
        }
        if writefile then
            pcall(function() writefile(fileTarget, HttpService:JSONEncode(data)) end)
        end
        NameInput.Text = ""
        rebuildDropdownRows()
    end)

    UI:CreateButton("Settings", "Load Selected Profile", function()
        if selectedProfileName == "" then return end
        local fileTarget = "Kittylol_Profile_" .. selectedProfileName .. ".json"
        if isfile and isfile(fileTarget) then
            local ok, data = pcall(function() return HttpService:JSONDecode(readfile(fileTarget)) end)
            if ok and data then
                Config.AutoParry = data.AutoParry or true
                Config.ManualSpam = data.ManualSpam or false
                Config.Trigger = data.Trigger or false
                Config.SpamCPS = data.SpamCPS or 200
                Config.ParryAccuracy = data.ParryAccuracy or 3.3
                Config.AnimFix = data.AnimFix or true
                Config.CurveType = data.CurveType or "straight"
                Config.UiTransparency = data.UiTransparency or 0
                Config.BackgroundId = data.BackgroundId or "Background 1"

                if data.ThemeLeft then themeLeft = Color3.new(data.ThemeLeft[1], data.ThemeLeft[2], data.ThemeLeft[3]) end
                if data.ThemeRight then themeRight = Color3.new(data.ThemeRight[1], data.ThemeRight[2], data.ThemeRight[3]) end
                if data.TitleColorLeft then titleColorLeft = Color3.new(data.TitleColorLeft[1], data.TitleColorLeft[2], data.TitleColorLeft[3]) end
                if data.TitleColorRight then titleColorRight = Color3.new(data.TitleColorRight[1], data.TitleColorRight[2], data.TitleColorRight[3]) end

                MainFrame.BackgroundTransparency = Config.UiTransparency
                if BackgroundImage then
                    BackgroundImage.ImageTransparency = Config.UiTransparency
                    BackgroundImage.Image = GetBackgroundAsset(Config.BackgroundId)
                end
                UpdateGlobalTheme()
            end
        end
    end)

    UI:CreateButton("Settings", "Delete Selected Profile", function()
        if selectedProfileName == "" then return end
        local fileTarget = "Kittylol_Profile_" .. selectedProfileName .. ".json"
        if delfile and isfile and isfile(fileTarget) then
            pcall(function() delfile(fileTarget) end)
        end
        local currentList = getProfilesList()
        local index = table.find(currentList, selectedProfileName)
        if index then
            table.remove(currentList, index)
            saveProfilesList(currentList)
        end
        selectedProfileName = ""
        DropdownBtn.Text = "Select Profile v"
        rebuildDropdownRows()
    end)
end

local TabButtons = {}
local function AddTab(name)
    local _ = CreateTabFrame(name)
    local TabBtn = Instance.new("TextButton")
    local UICorner_Tab = Instance.new("UICorner")

    TabBtn.Parent = LeftMenu
    TabBtn.Size = UDim2.new(1, -12, 0, 28)
    TabBtn.Font = Enum.Font.GothamSemibold
    TabBtn.Text = "   " .. name
    TabBtn.TextXAlignment = Enum.TextXAlignment.Left
    TabBtn.TextSize = 11
    TabBtn.ZIndex = 2
    TabBtn.BorderSizePixel = 0

    UICorner_Tab.CornerRadius = UDim.new(0, 4)
    UICorner_Tab.Parent = TabBtn

    local function renderTabState()
        if not TabBtn then return end
        if CurrentTab == name then
            TabBtn.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
            TabBtn.BackgroundTransparency = 0.88
            TabBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
        else
            TabBtn.BackgroundTransparency = 1
            TabBtn.TextColor3 = Color3.fromRGB(150, 150, 160)
        end
    end

    TabBtn.MouseButton1Click:Connect(function()
        CurrentTab = name
        for tName, frame in pairs(TabContainers) do
            if frame then frame.Visible = (tName == CurrentTab) end
        end
        for _, btnUpdate in pairs(TabButtons) do
            pcall(btnUpdate)
        end
    end)

    table.insert(TabButtons, renderTabState)
    renderTabState()
end

local function RollLootboxReward(lootbox_name, weapon_name)
    local reward_info = modules.CosmeticLibrary.Rewards[lootbox_name]
    if not reward_info or not reward_info.GetContents then return nil end

    local ok, contents = pcall(function() return reward_info:GetContents() end)
    if not ok or not contents or #contents == 0 then return nil end

    local inventory = modules.PlayerDataController:Get("CosmeticInventory")
    local available_rewards = {}

    for _, cosmetic_name in ipairs(contents) do
        if not modules.CosmeticLibrary:OwnsCosmetic(inventory, cosmetic_name) then
            table.insert(available_rewards, cosmetic_name)
        end
    end

    if #available_rewards == 0 then
        available_rewards = contents
    end

    local rolled_name = available_rewards[math.random(#available_rewards)]

    local cosmetic = modules.CosmeticLibrary.Cosmetics[rolled_name]
    local is_weapon_cosmetic =
        cosmetic
        and modules.CosmeticLibrary.Types
        and modules.CosmeticLibrary.Types[cosmetic.Type]
        and modules.CosmeticLibrary.Types[cosmetic.Type].IsWeaponCosmetic

    return {
        Name = rolled_name,
        Quantity = 1,
        Weapon = is_weapon_cosmetic and weapon_name or nil,
        ExpireTime = nil,
    }
end

        if char and char:FindFirstChild("HumanoidRootPart") then

                            owned[w.Name] = true

                            table.insert(inventory_state.fake_weapon_owned or {}, {
                                Name = weapon_name,
                                Level = 1,

                            modules.LootboxEffect(pos, lootbox_name, {{ RewardData = reward_data }}, delay, nil, LocalPlayer)

local function AddToBackpack()

    local new_entry = inventory_state.specific.new_entry
    local quantity = inventory_state.specific.quantity
    local weapon_name = inventory_state.specific.weapon_name

    if not modules.CosmeticLibrary or not modules.CosmeticLibrary.Rewards then return end
    if not modules.CosmeticLibrary.Rewards[item_name] then return end

    if not weapon_name or weapon_name == "" then
        weapon_name = "IsRandom"
    end

    local reward_entry = {
        Name = item_name,
        Quantity = quantity or 1,
        Weapon = weapon_name,
        ExpireTime = nil
    }

    if new_entry then
        table.insert(inventory_state.fake_owned, reward_entry)
    else
        local found = false
        for _, entry in ipairs(inventory_state.fake_owned) do
            if entry.Name == reward_entry.Name and entry.Weapon == reward_entry.Weapon and entry.Quantity < 99 then
                entry.Quantity = (entry.Quantity or 1) + (reward_entry.Quantity or 1)
                found = true
                break
            end
        end
        if not found then
            table.insert(inventory_state.fake_owned, reward_entry)
        end
    end

    ReplicateInventory()
end

                            function ColorpickerOptions:Set(color, trans, ignore)
                                if not ColorpickerOptions.Transparency then

                                    Colorpicker.HuePosition = Hue
                                    HuePicker.Position = UDim2New(0, -2, 1 - Hue, -2)
                                end
                                pcall(ColorpickerOptions.Callback, FromRGBA(color.R * 255, color.G * 255, color.B * 255), trans)

                            end

                                        Unclick = Unclick,
                                        Objects = Objects,
                                        Selected = false,
                                    }
                        
                                    theme_event.Event:Connect(function ()
                                        if Dropdown["FValues"][name].Selected then
                                            DBName.TextColor3 = theme.accent
                                        end
                                    end)
                                end
                        
                                return Dropdown
                            end
                        
                            function Dropdown:Update()
                                for _, v in pairs(Dropdown.FValues) do
                                    if Dropdown.FValue == _ then
                                        v.Click()
                                    else
                                        v.Unclick()
                                    end
                                end
                        
                                return Dropdown
                            end
                        
                            function Dropdown:Display()
                                if DropdownOptions.Multi then
                                    local CurrentText = {}
                        
                                    if #Dropdown.FValue > 0 then
                                        for _,v in pairs(Dropdown.FValue) do
                                            CurrentText[#CurrentText + 1] = v
                        
                                            local Text = table.concat(CurrentText, ", ")
                                            DropdownText.Text = Text
                                        end
                                    else
                                        DropdownText.Text = "-"
                                    end
                                else
                                    DropdownText.Text = Dropdown.FValue ~= "" and Dropdown.FValue or "-"
                                end
                        
                                return Dropdown
                            end
                        
                            function Dropdown:Set(value)
                                if DropdownOptions.Multi then
                                    if typeof(value) == "table" then
                                        for _,v in pairs(value) do
                                            if not table.find(Dropdown.FValue, _) then
                                                Dropdown:Set(v)
                                            end
                                        end
                        
                                        local RemovedButtons = {}
                        
                                        for _,v in pairs(Dropdown.FValue) do
                                            if not table.find(value, _) then
                                                RemovedButtons[#RemovedButtons + 1] = v
                                            end
                                        end
                        
                                        pcall(DropdownOptions.Callback, Dropdown.FValue)
                                        UserInterface.Flags[DropdownOptions.Flag] = Dropdown.FValue
                                        UserInterface.Flags[DropdownOptions.Flag .. "f"] = { [1] = function(value) Dropdown:Refresh(value) end, [2] = function(value) Dropdown:Set(value) end }
                        
                                        return
                                    end
                        
                                    local Index = table.find(Dropdown.FValue, value)
                        
                                    if Index then
                                        table.remove(Dropdown.FValue, Index)
                        
                                        Dropdown:Display()
                        
                                        Dropdown.FValues[value].Unclick()
                        
                                        pcall(DropdownOptions.Callback, Dropdown.FValue)
                                        UserInterface.Flags[DropdownOptions.Flag] = Dropdown.FValue
                                        UserInterface.Flags[DropdownOptions.Flag .. "f"] = { [1] = function(value) Dropdown:Refresh(value) end, [2] = function(value) Dropdown:Set(value) end }
                                    else
                                        Dropdown.FValue[#Dropdown.FValue + 1] = value
                        
                                        Dropdown:Display()
                        
                                        Dropdown.FValues[value].Click()
                        
                                        pcall(DropdownOptions.Callback, Dropdown.FValue)
                                        UserInterface.Flags[DropdownOptions.Flag] = Dropdown.FValue
                                        UserInterface.Flags[DropdownOptions.Flag .. "f"] = { [1] = function(value) Dropdown:Refresh(value) end, [2] = function(value) Dropdown:Set(value) end }
                                    end
                                else
                                    Dropdown.FValue = value
                        
                                    self:Update()
                        
                                    Dropdown:Display()
                                    pcall(DropdownOptions.Callback, Dropdown.FValue)
                                    UserInterface.Flags[DropdownOptions.Flag] = Dropdown.FValue
                                    UserInterface.Flags[DropdownOptions.Flag .. "f"] = { [1] = function(value) Dropdown:Refresh(value) end, [2] = function(value) Dropdown:Set(value) end }
                                end
                        
                                return Dropdown
                            end

                                for _,v in pairs(Dropdown.FValues) do
                                    v.Objects.Name:Destroy()
                                    v = nil
                                end
                        
                                Inline.Size = UDim2New(0, 229, 0, 0)

                                if DropdownOptions.Multi then
                                    table.clear(Dropdown.FValue)
                                    Count = 0
                        
                                    for _,v in pairs(tbl) do
                                        Dropdown:CreateValue(v)
                                    end
                        
                                    Dropdown:Display()
                        
                                    pcall(DropdownOptions.Callback, Dropdown.FValue)
                                    UserInterface.Flags[DropdownOptions.Flag] = Dropdown.FValue
                                    UserInterface.Flags[DropdownOptions.Flag .. "f"] = { [1] = function(value) Dropdown:Refresh(value) end, [2] = function(value) Dropdown:Set(value) end }
                                else
                                    Count = 0
                        
                                    for _,v in pairs(tbl) do
                                        Dropdown:CreateValue(v)
                                    end
                        
                                    Dropdown.FValue = nil
                        
                                    Dropdown:Update()
                                    Dropdown:Display()
                        
                                    pcall(DropdownOptions.Callback, Dropdown.FValue)
                                    UserInterface.Flags[DropdownOptions.Flag] = Dropdown.FValue
                                    UserInterface.Flags[DropdownOptions.Flag .. "f"] = { [1] = function(value) Dropdown:Refresh(value) end, [2] = function(value) Dropdown:Set(value) end }
                                end
                        
                                for _, v in pairs(tbl) do
                                    Dropdown:CreateValue(v)
                                end
                            
                                Dropdown:Set(DropdownOptions.Default)
                        
                                return Dropdown

                            end

            weapon = entry.Weapon,
        })
    end
    return list
end

local selectedOpenEntry = nil
local openQuantity = 1

local openEntryDropdown, _ = openSection:Dropdown({
    title = "Backpack Entry to Open",
    values = (function()
        local names = {}
        for _, v in ipairs(GetOpenableList()) do
            table.insert(names, v.display)
        end
        if #names == 0 then names = {"None"} end
        return names
    end)(),
    default = "None",
    callback = function(v)
        for _, entry in ipairs(GetOpenableList()) do
            if entry.display == v then
                selectedOpenEntry = entry
                openQuantity = entry.quantity or 1
                break
            end
        end
    end
})

local function RefreshOpenDropdown()
    local list = GetOpenableList()
    local displayNames = {}
    for _, v in ipairs(list) do
        table.insert(displayNames, v.display)
    end
    if #displayNames == 0 then displayNames = {"None"} end
    openEntryDropdown:Refresh(displayNames, displayNames[1] or "None")
    selectedOpenEntry = nil
    openQuantity = 1
end

openSection:Slider({
    title = "Quantity to Open",
    min = 1,
    max = 99,
    default = 1,
    suffix = "",
    callback = function(v)
        openQuantity = v
    end
})

openSection:Button({
    title = "Open Selected",
    callback = function()
        if not selectedOpenEntry or not selectedOpenEntry.index then
            print("[kmord] No backpack entry selected!")
            return
        end
        UseUnclaimedReward(selectedOpenEntry.index, openQuantity)
        RefreshOpenDropdown()
        print("[kmord] Opened: " .. tostring(selectedOpenEntry.name))
    end
})

openSection:Button({
    title = "Refresh List",
    callback = function()
        RefreshOpenDropdown()
        print("[kmord] Refreshed open list.")
    end
})


-- ============================================================
-- AVATAR PREVIEW TAB (ViewportFrame always on the right)
-- ============================================================
local previewTab = Interface:Tab("preview")

local previewSection = previewTab:Section("Avatar Clone", "left")

local previewState = {
    clone = nil,
    worldModel = nil,
    viewport = nil,
    camera = nil,
    yaw = 0,
    dragging = false,
    dragStartX = 0,
    basePivot = nil,
    holder = nil,
}

local function DestroyPreview()
    if previewState.clone then
        pcall(function() previewState.clone:Destroy() end)
        previewState.clone = nil
    end
    if previewState.holder then
        pcall(function() previewState.holder:Destroy() end)
        previewState.holder = nil
        previewState.viewport = nil
        previewState.worldModel = nil
        previewState.camera = nil
    end
end

local function CreateAvatarPreview()
    DestroyPreview()

    local char = LocalPlayer.Character
    if not char or not char:FindFirstChildOfClass("Humanoid") then
        print("[kmord] No character to clone")
        return
    end

    local parentColumn = previewTab.Right
    if not parentColumn then
        print("[kmord] Could not find right column")
        return
    end

    local holder = Instance.new("Frame")
    holder.Name = "AvatarPreviewHolder"
    holder.Size = UDim2.new(0, 247, 0, 300)
    holder.BackgroundColor3 = Color3.fromRGB(12, 12, 12)
    holder.BorderSizePixel = 0
    holder.Parent = parentColumn

    local holderCorner = Instance.new("UICorner")
    holderCorner.CornerRadius = UDim.new(0, 6)
    holderCorner.Parent = holder

    local holderStroke = Instance.new("UIStroke")
    holderStroke.Color = Color3.fromRGB(40, 40, 40)
    holderStroke.Thickness = 1
    holderStroke.Parent = holder

    local title = Instance.new("TextLabel")
    title.Size = UDim2.new(1, -12, 0, 18)
    title.Position = UDim2.new(0, 6, 0, 4)
    title.BackgroundTransparency = 1
    title.Text = "Avatar Preview (drag to rotate)"
    title.TextColor3 = Color3.fromRGB(200, 200, 200)
    title.TextSize = 13
    title.Font = Enum.Font.SourceSans
    title.TextXAlignment = Enum.TextXAlignment.Left
    title.Parent = holder

    local viewport = Instance.new("ViewportFrame")
    viewport.Name = "Viewport"
    viewport.Size = UDim2.new(1, -12, 1, -28)
    viewport.Position = UDim2.new(0, 6, 0, 24)
    viewport.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
    viewport.BorderSizePixel = 0
    viewport.Parent = holder

    local cam = Instance.new("Camera")
    cam.Parent = viewport
    viewport.CurrentCamera = cam

    local worldModel = Instance.new("WorldModel")
    worldModel.Parent = viewport

    -- Clone character, face straight forward (toward camera)
    char.Archivable = true
    local clone = char:Clone()
    clone.Name = "PreviewClone"
    for _, d in ipairs(clone:GetDescendants()) do
        if d:IsA("Script") or d:IsA("LocalScript") or d:IsA("Tool") or d:IsA("Sound") then
            d:Destroy()
        end
    end

    local hrp = clone:FindFirstChild("HumanoidRootPart") or clone.PrimaryPart or clone:FindFirstChildWhichIsA("BasePart")
    if hrp then
        clone:PivotTo(CFrame.new(0, 0, 0) * CFrame.Angles(0, math.rad(180), 0))
        clone.PrimaryPart = hrp
        local hum = clone:FindFirstChildOfClass("Humanoid")
        if hum then
            hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
            pcall(function() hum:ChangeState(Enum.HumanoidStateType.GettingUp) end)
        end
    end
    clone.Parent = worldModel

    local pivot = clone:GetPivot()
    cam.CFrame = CFrame.new(pivot.Position + Vector3.new(0, 1.4, 5.5), pivot.Position + Vector3.new(0, 1.1, 0))

    previewState.clone = clone
    previewState.worldModel = worldModel
    previewState.viewport = viewport
    previewState.camera = cam
    previewState.basePivot = clone:GetPivot()
    previewState.yaw = 0
    previewState.holder = holder

    local function setYaw(yaw)
        previewState.yaw = yaw
        if previewState.clone and previewState.basePivot then
            previewState.clone:PivotTo(previewState.basePivot * CFrame.Angles(0, math.rad(yaw), 0))
        end
    end

    viewport.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            previewState.dragging = true
            previewState.dragStartX = input.Position.X
        end
    end)
    viewport.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            previewState.dragging = false
        end
    end)
    inputService.InputChanged:Connect(function(input)
        if previewState.dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position.X - previewState.dragStartX
            previewState.dragStartX = input.Position.X
            setYaw(previewState.yaw + delta * 0.5)
        end
    end)

    task.defer(function()
        if previewTab.Right then
            previewTab.Right.CanvasSize = UDim2.new(0, 0, 0, previewTab.Right.AbsoluteCanvasSize.Y + 320)
        end
    end)

    print("[kmord] Avatar preview ready (facing forward)")
end

        end
    end

    end

-- // FLAWLESS TARGETING (Bypasses Broken Team Logic)
local GlobalTarget = nil
local function UpdateTarget()
    local dist = math.huge
    local target = nil
    local myHRP = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    
    if not myHRP then 
        GlobalTarget = nil 
        return 
    end

    for _, v in ipairs(Players:GetPlayers()) do
        if v ~= LocalPlayer and v.Character then
            local targetHRP = v.Character:FindFirstChild("HumanoidRootPart")
            local hum = v.Character:FindFirstChild("Humanoid")
            
            -- Checks if alive and exists
            if targetHRP and hum and hum.Health > 0 then
                local mag = (targetHRP.Position - myHRP.Position).Magnitude
                if mag < dist then
                    dist = mag
                    target = v.Character
                end
            end
        end
    end
    GlobalTarget = target
end

-- // WALLBANG REDIRECT (Mobile Safe Hook)
local oldNamecall
oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
    local method = getnamecallmethod()
    local args = {...}
    
    if not checkcaller() and Config.Wallbang and GlobalTarget then
        if method == "Raycast" or method == "FindPartOnRay" then
            local head = GlobalTarget:FindFirstChild("Head")
            if head then
                if method == "Raycast" then
                    args[2] = (head.Position - args[1]).Unit * 15000
                else
                    args[1] = Ray.new(args[1].Origin, (head.Position - args[1].Origin).Unit * 15000)
                end
                return oldNamecall(self, unpack(args))
            end
        end
    end
    return oldNamecall(self, ...)
end)

-- // APEX MOVEMENT & COMBAT (Runs Before Physics)
local orbitAngle = 0
RunService.Stepped:Connect(function()
    UpdateTarget()
    
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    
    if hrp then
        -- Noclip Logic
        if Config.Noclip or Config.TeleportAura then
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") and part.CanCollide then
                    part.CanCollide = false
                end
            end
        end

        -- Teleport Aura Logic
        if Config.TeleportAura and GlobalTarget and GlobalTarget:FindFirstChild("HumanoidRootPart") then
            local targetHRP = GlobalTarget.HumanoidRootPart
            orbitAngle = orbitAngle + math.rad(Config.AuraSpeed * 0.05)
            
            local offset = CFrame.Angles(0, orbitAngle, 0) * CFrame.new(0, 0, Config.AuraDistance)
            hrp.CFrame = targetHRP.CFrame * offset
            
            -- Kill all momentum so the server doesn't snap you back
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
            
            -- Mobile Auto-Shoot Integration
            if Config.RapidFire then
                local tool = char:FindFirstChildOfClass("Tool")
                if tool then
                    tool:Activate()
                end
            end
        end
    end
end)

-- =============================================
--   UI COMPATIBILITY LAYER
--   The Allusive library uses:
--     main:create_tab(...)
--     tab:create_module(...)
--     module:create_checkbox/create_feature/create_slider/create_dropdown
--   The old code used a different API, so the old wrappers silently
--   returned nil and none of the tabs/features were created.
-- =============================================

local function Tab(name)
    local ok, tab = pcall(function()
        return main:create_tab(name, "rbxassetid://121663636062758")
    end)
    if not ok or not tab then return nil end

    -- Put all controls of the tab in one opened module so they are visible.
    local ok2, module = pcall(function()
        return tab:create_module({
            title = name,
            description = "Kittylol",
            section = "left"
        })
    end)
    if ok2 and module then
        pcall(function() module:change_state(true) end)
        tab._kitty_module = module
    end
    return tab
end

local function getModule(tab)
    return tab and tab._kitty_module
end

local function Section(tab, name)
    local m = getModule(tab)
    if not m then return tab end
    pcall(function()
        m:create_divider({title=name, showtopic=true})
    end)
    return m
end

local function Toggle(tab, flag, text, cb)
    local m = getModule(tab)
    if not m then return nil end
    return m:create_checkbox({
        title=text,
        flag=flag,
        callback=cb or function() end
    })
end

local function Button(tab, flag, text, cb)
    local m = getModule(tab)
    if not m then return nil end
    return m:create_feature({
        title=text,
        flag=flag,
        default="Unknown",
        disablecheck=true,
        button_callback=cb or function() end
    })
end

local function Slider(tab, flag, text, min, max, def, suffix, cb)
    local m = getModule(tab)
    if not m then return nil end
    return m:create_slider({
        title=text .. (suffix and (" ["..suffix.."]") or ""),
        flag=flag,
        minimum_value=min,
        maximum_value=max,
        value=def,
        round_number=true,
        callback=cb or function() end
    })
end

local function Dropdown(tab, flag, text, values, cb, defIdx)
    local m = getModule(tab)
    if not m then return nil end
    local settings = {
        title=text,
        flag=flag,
        options=values or {},
        maximum_options=math.max(#(values or {}), 1),
        multi_dropdown=false,
        callback=cb or function() end
    }
    local manager = m:create_dropdown(settings)
    return {
        _manager=manager,
        _settings=settings,
        SetValues=function(self, newValues)
            newValues = newValues or {}
            self._settings.options = newValues
            self._settings.maximum_options = math.max(#newValues, 1)
            local ok, result = pcall(function()
                return self._manager:New(self._settings)
            end)
            if ok and result then self._manager=result end
            return self
        end,
        SetValue=function(self, value)
            if self._manager and self._manager.update then
                pcall(function() self._manager:update(value) end)
            end
            return self
        end
    }
end

local function ColorPicker(tab, flag, text, def, cb)
    -- This Allusive build has no ColorPicker API. Keep the setting functional
    -- by showing its current value instead of throwing an error.
    local m = getModule(tab)
    if not m then return nil end
    return m:create_text({
        text=text .. ": " .. tostring(def),
        customScale=28
    })
end

    return m:create_divider({

        disableline=true

end

local function Divider(tab)

end

local function DropdownSetValues(dd, values)

local vpRp = RaycastParams.new(); vpRp.FilterType = Enum.RaycastFilterType.Exclude
local function isVis(orig, part)
    if not part then return false end
    local ok, res = pcall(function()
        vpRp.FilterDescendantsInstances = {lp.Character, part.Parent}
        local h = ws:Raycast(orig, part.Position - orig, vpRp)
        return h == nil or h.Instance:IsDescendantOf(part.Parent)
    end)
    return ok and res or false
end

local function inFov(part, sz)
    if not part then return false end
    local sp, on = cam:WorldToViewportPoint(part.Position)
    if not on then return false end
    return (Vector2.new(sp.X,sp.Y) - getScr()).Magnitude <= sz
end

local function angDiff(f, t)
    local d = t - f
    return ((d + math.pi) % (math.pi*2)) - math.pi
end

local function abEffFov() return AB.dynFov and AB.fov*(cam.FieldOfView/70) or AB.fov end

local function abTgtValid(pt)
    if not pt or not pt.Parent then return false end
    local h = pt.Parent:FindFirstChildOfClass("Humanoid")
    if not h or h.Health<=0 then return false end
    if not inFov(pt, abEffFov()) then return false end
    if AB.vis and not isVis(cam.CFrame.Position, pt) then return false end
    if AB.team then
        local p = Players:GetPlayerFromCharacter(pt.Parent)
        if p and sameTm(p) then return false end
    end
    return true
end

local function abFind()
    local center = AB.useCenter and getScr() or getMousePos()
    local fv = abEffFov()
    local best, bd = nil, fv
    for _, p in ipairs(Players:GetPlayers()) do
        if p == lp then continue end
        if AB.team and sameTm(p) then continue end
        local chr = p.Character; if not chr then continue end
        local hm = chr:FindFirstChildOfClass("Humanoid")
        if not hm or hm.Health<=0 then continue end
        local pt = bPart(chr, AB.part); if not pt then continue end
        local sp, on = cam:WorldToViewportPoint(pt.Position)
        if not on then continue end
        local d = (Vector2.new(sp.X,sp.Y)-center).Magnitude
        if d < bd then
            if AB.vis and not isVis(cam.CFrame.Position, pt) then continue end
            bd = d; best = pt
        end
    end
    return best
end

local function getPredictedPos(pt)
    local now, curPos = tick(), pt.Position
    local result = curPos
    if abPrevPos and (now - abPrevTick) > 0.001 then
        local vel = (curPos - abPrevPos) / (now - abPrevTick)
        if vel.Magnitude < 200 then result = curPos + vel * (0.08 * AB.speedAttack) end
    end
    abPrevPos, abPrevTick = curPos, now
    return result
end

local function applySmartLock(tRot, cur, dt, sd, fovR)
    local df = AB.speed
    if fovR > 0 then
        local n = math.clamp(sd / fovR, 0, 1)
        df = AB.speed + (AB.speedScaleFov / 100) * (1 - n) * math.max(AB.speedAttack - AB.speed, 0)
    end
    local eff = math.clamp(df, 0.01, 5)
    local fac = math.clamp(eff * dt * 6, 0, 1)
    if AB.lockMode == "Soft" or AB.lockMode == "Magnet" then
        return Vector2.new(cur.X + (tRot.X - cur.X)*fac, cur.Y + angDiff(cur.Y, tRot.Y)*fac)
    else
        local needNew = (abCurveStart == nil) or (abCurveEnd == nil)
        if not needNew then
            local dx = tRot.X - abCurveEnd.X
            local dy = angDiff(abCurveEnd.Y, tRot.Y)
            if math.sqrt(dx*dx + dy*dy) > 1.2 then needNew = true end
        end
        if needNew then abCurveStart = cur; abCurveProgress = 0 end
        abCurveEnd = tRot
        abCurveProgress = math.clamp(abCurveProgress + (AB.curvePct/100)*3*eff*dt, 0, 1)
        local cv = abCurveProgress^2 * (3 - 2*abCurveProgress)
        local nP = abCurveStart.X + (abCurveEnd.X - abCurveStart.X) * cv
        local nY = abCurveStart.Y + angDiff(abCurveStart.Y, abCurveEnd.Y) * cv
        if abCurveProgress >= 1 then abCurveStart = Vector2.new(nP, nY); abCurveProgress = 0 end
        return Vector2.new(nP, nY)
    end
end

local function buildSilentWrapper(originalFunc)
    return function(self, origin, direction, distance, ...)
        if silentCfg.enabled and math.random(1,100)<=silentCfg.hitChance then
            local myRoot = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
            local orig = myRoot and myRoot.Position or cam.CFrame.Position

                    pcall(function()
                        local hrp = target.Parent and target.Parent:FindFirstChild("HumanoidRootPart")
                        local vel = hrp and hrp.AssemblyLinearVelocity or Vector3.zero
                        pp = target.Position + vel*(silentCfg.prediction*0.05)
                    end)
                end
                return originalFunc(self, origin, pp-origin, distance, ...)
            end
        end
        return originalFunc(self, origin, direction, distance, ...)
    end
end

local function hookSilent()
    if silentHooked or not util then return end
    local ok, mt = pcall(getrawmetatable, util)
    if ok and mt then
        origMeta = mt; origIndex = rawget(mt, "__index")
        rawset(mt, "__index", function(t, k)
            if k == "Raycast" then
                local rf
                if type(origIndex)=="function" then rf=origIndex(t,k)
                elseif type(origIndex)=="table" then rf=origIndex[k]
                else rf=rawget(t,k) end
                if rf then return buildSilentWrapper(rf) end
            end
            if type(origIndex)=="function" then return origIndex(t,k)
            elseif type(origIndex)=="table" then return origIndex[k] end

        end)

    end
    origRaycast = rawget(util,"Raycast") or util.Raycast
    if origRaycast then rawset(util,"Raycast",buildSilentWrapper(origRaycast)); silentHooked=true end
end

    elseif origRaycast and util then pcall(function() rawset(util,"Raycast",origRaycast) end); origRaycast=nil end

end

local function rageCalcPoint(origin,targetPos,targetChar)
    ray_params.FilterDescendantsInstances={lp.Character,targetChar}
    if not ws:Raycast(origin,targetPos-origin,ray_params) then return origin,nil end
    for _,offset in next,rageCfg.vectors do
        local sp = origin+offset
        if not ws:Raycast(sp,targetPos-sp,ray_params) then return sp,offset.Y end
    end
    return nil,nil
end

local function improvedCalcPoint(origin,targetPos,targetChar)
    ray_params.FilterDescendantsInstances={lp.Character,targetChar}
    local function tryPoint(sp)
        local r = ws:Raycast(sp,targetPos-sp,ray_params)
        return r==nil or r.Instance:IsDescendantOf(targetChar)
    end
    if tryPoint(origin) then return origin,nil end
    for _,offset in next,rageCfg.vectors do
        local sp = origin+offset
        if tryPoint(sp) then return sp,offset.Y end
    end
    if rageCfg.improv.multiPass then
        for _,offset in next,rageCfg.vectors do
            for angle=-45,45,rageCfg.improv.scanAngle do
                local rad = math.rad(angle)
                local sp = origin+Vector3.new(offset.X*math.cos(rad)-offset.Z*math.sin(rad),offset.Y,offset.X*math.sin(rad)+offset.Z*math.cos(rad))
                if tryPoint(sp) then return sp,offset.Y end
            end
        end
    end
    if rageCfg.improv.spiralScan then
        for layer=1,4 do local h=layer*8
            for s=0,11 do
                local a=(s/12)*math.pi*2
                local sp = origin+Vector3.new(math.cos(a)*2*layer,h,math.sin(a)*2*layer)
                if tryPoint(sp) then return sp,h end
            end
        end
    end
    local sides = {Vector3.new(2,0,0),Vector3.new(-2,0,0),Vector3.new(0,0,2),Vector3.new(0,0,-2),Vector3.new(3,8,0),Vector3.new(-3,8,0),Vector3.new(0,8,3),Vector3.new(0,8,-3)}
    for _,side in next,sides do
        local sp = origin+side
        if tryPoint(sp) then return sp,side.Y end
    end
    if rageCfg.improv.adaptDist then
        local scale = math.clamp((origin-targetPos).Magnitude/50,0.5,3)
        for _,offset in next,rageCfg.vectors do
            local sc = Vector3.new(offset.X*scale,offset.Y*scale,offset.Z*scale)
            local sp = origin+sc
            if tryPoint(sp) then return sp,sc.Y end
        end
    end
    return nil,nil
end

        end

        if self.conn2 then self.conn2:Disconnect();self.conn2=nil end
        if self.task1 then task.cancel(self.task1);self.task1=nil end

    local minX,minY,maxX,maxY,any = math.huge,math.huge,-math.huge,-math.huge,false
    for _,corner in ipairs(corners) do
        local sp,onScreen = cam:WorldToViewportPoint(corner)
        if onScreen and sp.Z>0 then
            any=true
            if sp.X<minX then minX=sp.X end
            if sp.Y<minY then minY=sp.Y end
            if sp.X>maxX then maxX=sp.X end
            if sp.Y>maxY then maxY=sp.Y end
        end
    end
    if not any then return nil end
    local vx,vy = cam.ViewportSize.X, cam.ViewportSize.Y
    minX=math.clamp(minX,0,vx);minY=math.clamp(minY,0,vy);maxX=math.clamp(maxX,0,vx);maxY=math.clamp(maxY,0,vy)
    local w,h = maxX-minX, maxY-minY

    return minX,minY,w,h
end

        if not headOn or headSP.Z<=0 then hideObj(obj); continue end

        if espCfg.visibleOnly and not visible then hideObj(obj); continue end

        if not bx then hideObj(obj); continue end

        if espCfg.box and not espCfg.corners then
            setLine(obj.bT,bx,by,bx+bw,by,boxCol);setLine(obj.bB,bx,by+bh,bx+bw,by+bh,boxCol)

            obj.cTLh.Visible=false;obj.cTLv.Visible=false;obj.cTRh.Visible=false;obj.cTRv.Visible=false
            obj.cBLh.Visible=false;obj.cBLv.Visible=false;obj.cBRh.Visible=false;obj.cBRv.Visible=false
        elseif espCfg.corners then
            obj.bT.Visible=false;obj.bB.Visible=false;obj.bL.Visible=false;obj.bR.Visible=false

            setLine(obj.cTRh,bx+bw-cl,by,bx+bw,by,boxCol);setLine(obj.cTRv,bx+bw,by,bx+bw,by+cl,boxCol)
            setLine(obj.cBLh,bx,by+bh,bx+cl,by+bh,boxCol);setLine(obj.cBLv,bx,by+bh-cl,bx,by+bh,boxCol)
            setLine(obj.cBRh,bx+bw-cl,by+bh,bx+bw,by+bh,boxCol);setLine(obj.cBRv,bx+bw,by+bh-cl,bx+bw,by+bh,boxCol)
        else
            obj.bT.Visible=false;obj.bB.Visible=false;obj.bL.Visible=false;obj.bR.Visible=false
            obj.cTLh.Visible=false;obj.cTLv.Visible=false;obj.cTRh.Visible=false;obj.cTRv.Visible=false

            end
            for i=#bDefs+1,MAX_BONES do if obj.bones[i] then obj.bones[i].Visible=false end end
        else for _,bone in ipairs(obj.bones) do bone.Visible=false end end

        if espCfg.name then
            obj.nameTag.Text=p.DisplayName~="" and p.DisplayName or p.Name
            obj.nameTag.Size=espCfg.nameSize;obj.nameTag.Color=espCfg.nameColor
            obj.nameTag.Position=Vector2.new(bx+bw/2,by-espCfg.nameSize-4);obj.nameTag.Visible=true
        else obj.nameTag.Visible=false end

        if espCfg.weapon then
            local nameOff = espCfg.name and (espCfg.nameSize+2) or 0
            obj.weaponTag.Text="["..getWeapon(char).."]"
            obj.weaponTag.Size=espCfg.weaponSize;obj.weaponTag.Color=espCfg.weaponColor
            obj.weaponTag.Position=Vector2.new(bx+bw/2,by-espCfg.weaponSize-4-nameOff);obj.weaponTag.Visible=true

        else obj.distTag.Visible=false end

            if s=="Left" then local x=bx-bO-bW;bgF=Vector2.new(x,by);bgT=Vector2.new(x,by+bh);fgF=Vector2.new(x,by+bh);fgT=Vector2.new(x,by+bh-bh*pct)

        else obj.hpBg.Visible=false;obj.hpFg.Visible=false end
    end
    for p in pairs(espPool) do if not espValidSet[p] then destroyESPObjects(p) end end
end

local function buildWeaponList()
    local list = {}
    for _,t in ipairs({PRIMARY_WEAPONS,SECONDARY_WEAPONS,MELEE_WEAPONS,UTILITY_WEAPONS}) do
        for _,n in ipairs(t) do table.insert(list,n) end
    end
    table.sort(list); return list
end

local function cosmeticIsForWeapon(cd, wn)
    if not cd or not wn then return false end
    if cd.Universal then return true end
    for _,f in ipairs({"Weapon","Item","WeaponName","ItemName"}) do
        local w = cd[f]
        if w then
            if type(w)=="string" and w==wn then return true end
            if type(w)=="table" then for _,n in pairs(w) do if tostring(n)==wn then return true end end end
        end
    end
    for _,f in ipairs({"Weapons","Items","WeaponNames","ItemNames","AppliesTo"}) do
        local w = cd[f]
        if w and type(w)=="table" then for _,n in pairs(w) do if tostring(n)==wn then return true end end end
    end
    return false
end

local function buildCosmeticList(cType, wn)
    local list = {"None"}
    if not CosmeticLibrary or not CosmeticLibrary.Cosmetics then return list end
    local matches,hasSpecific = {},false
    for name, data in pairs(CosmeticLibrary.Cosmetics) do
        if data and data.Type == cType then
            if wn then
                if cosmeticIsForWeapon(data, wn) then table.insert(matches, name); hasSpecific=true end
            else table.insert(matches, name) end
        end
    end
    if cType == "Skin" then table.sort(matches); for _,n in ipairs(matches) do table.insert(list,n) end; return list end
    if not hasSpecific and wn then
        for name, data in pairs(CosmeticLibrary.Cosmetics) do
            if data and data.Type == cType then table.insert(matches, name) end
        end
    end
 table.sort(matches); for _,n in ipairs(matches) do table.insert(list,n) end; return list
end

RunService.RenderStepped:Connect(function(dt)
    cam = ws.CurrentCamera

    pcall(renderESP)

    pcall(function()
        if not CameraController then return end
        if camCfg.fovChanger then CameraController._base_fov = camCfg.fovValue end
        if camCfg.noShake then CameraController._shake_enabled=false;CameraController.ShakeCFrame=CFrame.identity end
        if camCfg.noSway then CameraController._sway_spring.Value=Vector2.zero;CameraController._sway_spring.Target=Vector2.zero end
        if camCfg.noBobbing then CameraController._bobbing_speed_spring.Target=0;CameraController._bobbing_speed_spring.Value=0;CameraController._bobbing_value_spring.Target=0;CameraController._bobbing_value_spring.Value=0 end
        if camCfg.noJump then CameraController._jump_spring.Target=0;CameraController._jump_spring.Value=0 end
        if camCfg.noSlideTilt then CameraController._sliding_spring.Target=0;CameraController._sliding_spring.Value=0 end
        if camCfg.forceCrosshair then CameraController._crosshair_disabled=false end
        if camCfg.sensOverride then
            CameraController._camera_sensitivity=camCfg.sensValue/100
            CameraController._camera_sensitivity_ads_multiplier=camCfg.sensADS/100
            CameraController._camera_sensitivity_ads_multiplier_scoped=camCfg.sensScoped/100
            CameraController._camera_sensitivity_x=camCfg.sensX/100
            CameraController._camera_sensitivity_y=camCfg.sensY/100
        end
        if camCfg.vmEnabled then
            CameraController.ViewModelOffsetCFrame=CFrame.new(camCfg.vmX,camCfg.vmY,camCfg.vmZ)
        end
    end)

    -- Aimbot
    if AB.on and CameraController then
        if abDoReact then abReactT-=dt;if abReactT<=0 then abDoReact=false end end
        if not abDoReact then
            if abTgt and not abTgtValid(abTgt) then abTgt=nil;abLockT=0;abPrevPos=nil;abCurveStart=nil;abCurveEnd=nil;abCurveProgress=0 end
            if not abTgt then abTgt=abFind();if abTgt then abLockT=0;abDoReact=true;abReactT=AB.reactTime/1000;abPrevPos=nil;abCurveStart=nil;abCurveEnd=nil;abCurveProgress=0 end end
            if abTgt then
                abLockT+=dt
                if abLockT>AB.maxLock then abTgt=nil;abLockT=0;abPrevPos=nil;abCurveStart=nil;abCurveEnd=nil;abCurveProgress=0
                elseif inFov(abTgt,abEffFov()) then
                    pcall(function()
                        local aimPos=getPredictedPos(abTgt);local du=(aimPos-cam.CFrame.Position).Unit
                        local tRot=Vector2.new(math.asin(math.clamp(du.Y,-1,1)),math.atan2(-du.X,-du.Z))
                        local cur=CameraController.Rotation;local sp,_=cam:WorldToViewportPoint(aimPos)
                        local sd=(Vector2.new(sp.X,sp.Y)-getScr()).Magnitude
                        local newRot=applySmartLock(tRot,cur,dt,sd,abEffFov())
                        CameraController:ApplyRotationDelta(Vector2.new(newRot.X-cur.X,angDiff(cur.Y,newRot.Y)))
                    end)
                else abTgt=nil;abLockT=0;abPrevPos=nil;abCurveStart=nil;abCurveEnd=nil;abCurveProgress=0 end
            end
        end
    end

    -- FOV Circle
    if fovCfg.visible then
        local center=fovCfg.useCenter and getScr() or getMousePos()
        fovCircle.Visible=true;fovCircle.Color=fovCfg.color;fovCircle.Position=center
        local myRoot=lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
        local orig=myRoot and myRoot.Position or cam.CFrame.Position
        fovCircle.Radius=abTgt and(isVis(orig,abTgt) and fovCfg.onTarget or fovCfg.onBarrier) or fovCfg.fovSize
        if fovCfg.half then fovHalf.Visible=true;fovHalf.Color=fovCfg.color;fovHalf.Position=center;fovHalf.Radius=fovCircle.Radius*0.5
        else fovHalf.Visible=false end
        abCirc.Position=center;abCirc.Radius=abEffFov();abCirc.Visible=AB.on and AB.drawFov
    else fovCircle.Visible=false;fovHalf.Visible=false;abCirc.Visible=false end

    -- Silent FOV Circle
    if silentCfg.enabled and silentCfg.circleVisible then
        local center=silentCfg.useCenter and getScr() or getMousePos()
        silentCircle.Visible=true;silentCircle.Radius=silentCfg.fovSize
        silentCircle.Thickness=silentCfg.circleThick;silentCircle.Position=center
        local myRoot=lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
        local orig=myRoot and myRoot.Position or cam.CFrame.Position
        silentCircle.Color=getSilentTarget(orig) and silentCfg.targetColor or silentCfg.circleColor
    else silentCircle.Visible=false end
end)
